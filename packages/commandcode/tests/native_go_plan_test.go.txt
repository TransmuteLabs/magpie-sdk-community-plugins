var commandcodeGoDistribution = flag.String("commandcode-go-distribution", "", "")

func TestCommandCodeNativeGoProtocols(test *testing.T) {
	if *commandcodeGoDistribution == "" {
		test.Fatal("an installed native distribution is required")
	}
	fresh(test)
	test.Setenv("MAGPIE_NO_ACCOUNTS", "")
	test.Setenv("MAGPIE_ADDR", "")
	var mutex sync.Mutex
	var bodies [][]byte
	var sessions []string
	upstream := httptest.NewServer(http.HandlerFunc(func(writer http.ResponseWriter, request *http.Request) {
		switch request.URL.Path {
		case "/alpha/billing/subscriptions":
			io.WriteString(writer, `{"data":{"planId":"individual-go","status":"active"}}`)
		case "/alpha/billing/credits":
			io.WriteString(writer, `{"credits":{"monthlyCredits":9,"purchasedCredits":0,"freeCredits":0},"windowLimits":{"fiveHour":{"used":1,"cap":10}}}`)
		case "/provider/v1/models":
			io.WriteString(writer, `{"data":[{"id":"google/gemini-fixture","supported_endpoints":["/chat/completions"],"modalities":{"input":["text","image"]}}]}`)
		case "/alpha/generate":
			body, _ := io.ReadAll(request.Body)
			mutex.Lock()
			bodies = append(bodies, body)
			sessions = append(sessions, request.Header.Get("x-session-id"))
			attempt := len(bodies)
			mutex.Unlock()
			if request.Header.Get("Authorization") != "Bearer synthetic-go-key" ||
				request.Header.Get("x-command-code-version") != "1.89.1" ||
				request.Header.Get("x-cmd-zdr") != "1" {
				test.Error("native Go request lost selected account, harness identity or ZDR")
			}
			if bytes.Contains(body, []byte(`"$ref"`)) || !bytes.Contains(body, []byte("data:image/png;base64,AA==")) ||
				!bytes.Contains(body, []byte("another trigger")) {
				test.Errorf("client protocol lost its image or normalized tool constraints: %s", body)
			}
			writer.Header().Set("Content-Type", "application/x-ndjson")
			if attempt%2 == 1 {
				io.WriteString(writer, "{\"type\":\"text-delta\",\"text\":\"first\"}\n{\"type\":\"finish\",\"finishReason\":\"other\",\"rawFinishReason\":\"pause_turn\",\"totalUsage\":{\"inputTokens\":10,\"outputTokens\":2,\"inputTokenDetails\":{\"cacheReadTokens\":3}}}\n")
				return
			}
			io.WriteString(writer, "{\"type\":\"reasoning-delta\",\"text\":\"thought\"}\n{\"type\":\"text-delta\",\"text\":\"second\"}\n{\"type\":\"tool-call\",\"toolCallId\":\"native-tool\",\"toolName\":\"schedule\",\"input\":{\"on\":\"push\"}}\n")
			io.WriteString(writer, "{\"type\":\"finish\",\"finishReason\":\"tool-calls\",\"totalUsage\":{\"inputTokens\":7,\"outputTokens\":4,\"inputTokenDetails\":{\"cacheReadTokens\":1}}}\n")
		default:
			if request.Method == http.MethodGet && request.URL.Path == "/" && request.Header.Get("Authorization") == "" {
				writer.WriteHeader(http.StatusUnauthorized)
				return
			}
			test.Errorf("Go request used another transport: %s %s", request.Method, request.URL.Path)
			writer.WriteHeader(500)
		}
	}))
	defer upstream.Close()
	manager := extensionhost.Default()
	test.Cleanup(func() {
		ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
		defer cancel()
		if err := manager.Close(ctx); err != nil {
			test.Error(err)
		}
	})
	entry, err := manager.Install(test.Context(), *commandcodeGoDistribution)
	if err != nil {
		test.Fatal(err)
	}
	config, _ := json.Marshal(map[string]any{"apiURL": upstream.URL, "workingDirectory": "/synthetic/shop", "zdr": true})
	if _, err := manager.CompareAndSwapModuleConfig(test.Context(), entry.Manifest.ID, "0", config); err != nil {
		test.Fatal(err)
	}
	if _, err := manager.Store().SaveAccount(extensionhost.Account{Plugin: entry.Manifest.ID,
		Provider: "commandcode-native", ID: "synthetic-go-account", Auth: json.RawMessage(`{"apiKey":"synthetic-go-key"}`)}, 0); err != nil {
		test.Fatal(err)
	}
	provider.Changed()
	accounts, err := provider.ExtensionProviders()
	if err != nil || len(accounts) != 1 {
		test.Fatalf("native account missing: %+v %v", accounts, err)
	}
	models, err := accounts[0].Fetch(test.Context())
	if err != nil || len(models) != 2 || len(models[0].APIs) != 1 {
		test.Fatalf("Go catalog must select the generic Chat translation lane: %+v %v", models, err)
	}
	provider.Changed()
	server := New()
	test.Cleanup(server.idle.stop)
	id := provider.ExtensionProviderID(entry.Manifest.ID, "commandcode-native")
	model := id + "/google/gemini-fixture"
	schema := `{"type":"object","properties":{"trigger":{"type":"string","enum":["push","pull"]},"copy":{"$ref":"#/properties/trigger","description":"another trigger"}}}`
	cases := []struct{ path, body string }{
		{"/v1/chat/completions", `{"model":%q,"messages":[{"role":"user","content":[{"type":"text","text":"schedule"},{"type":"image_url","image_url":{"url":"data:image/png;base64,AA=="}}]}],"tools":[{"type":"function","function":{"name":"schedule","parameters":` + schema + `}}],"stream":%t}`},
		{"/v1/responses", `{"model":%q,"input":[{"role":"user","content":[{"type":"input_text","text":"schedule"},{"type":"input_image","image_url":"data:image/png;base64,AA=="}]}],"tools":[{"type":"function","name":"schedule","parameters":` + schema + `}],"stream":%t}`},
		{"/v1/messages", `{"model":%q,"max_tokens":1024,"messages":[{"role":"user","content":[{"type":"text","text":"schedule"},{"type":"image","source":{"type":"base64","media_type":"image/png","data":"AA=="}}]}],"tools":[{"name":"schedule","input_schema":` + schema + `}],"stream":%t}`},
	}
	operations := filepath.Join(manager.Namespace(), "private", entry.Manifest.ID, "operations", "*.json")
	priorOperations, err := filepath.Glob(operations)
	if err != nil {
		test.Fatal(err)
	}
	seenOperations := make(map[string]bool)
	for _, path := range priorOperations {
		seenOperations[path] = true
	}
	index := 0
	for _, streaming := range []bool{false, true} {
		for _, sample := range cases {
			body := fmt.Sprintf(sample.body, model, streaming)
			request := httptest.NewRequest(http.MethodPost, sample.path, strings.NewReader(body))
			request.RemoteAddr = "127.0.0.1:12345"
			request.Header.Set("Content-Type", "application/json")
			session := fmt.Sprintf("native-go-%d", index)
			request.Header.Set("X-Magpie-Session", session)
			reply := httptest.NewRecorder()
			server.Front().ServeHTTP(reply, request)
			if reply.Code != 200 || strings.Count(reply.Body.String(), "first") < 1 ||
				!strings.Contains(reply.Body.String(), "second") || !strings.Contains(reply.Body.String(), "schedule") {
				test.Fatalf("native Go protocol %s/stream=%t failed: %d %s", sample.path, streaming, reply.Code, reply.Body)
			}
			mutex.Lock()
			count := len(bodies)
			equal := count == (index+1)*2 && bytes.Equal(bodies[count-2], bodies[count-1]) &&
				sessions[count-2] != "" && sessions[count-2] == sessions[count-1]
			mutex.Unlock()
			if !equal {
				test.Fatalf("client protocol changed/retried the paused native request: %s calls=%d", sample.path, count)
			}
			paths, err := filepath.Glob(operations)
			if err != nil {
				test.Fatal(err)
			}
			var receipts []extensionhost.JournalEntry
			for _, path := range paths {
				if seenOperations[path] {
					continue
				}
				seenOperations[path] = true
				body, err := os.ReadFile(path)
				var receipt extensionhost.JournalEntry
				if err != nil || json.Unmarshal(body, &receipt) != nil {
					test.Fatal("native operation receipt is unreadable", err)
				}
				if receipt.Scope.Provider == "commandcode-native" && receipt.Scope.Account == "synthetic-go-account" {
					receipts = append(receipts, receipt)
				}
			}
			if len(receipts) != 1 || receipts[0].State != "completed" {
				test.Errorf("native Go protocol lost its completed SDK receipt: %s/stream=%t %+v", sample.path, streaming, receipts)
			}
			var rows []usage.Record
			for _, row := range usage.Load(time.Time{}) {
				if row.Session == session {
					rows = append(rows, row)
				}
			}
			if len(rows) != 1 || rows[0].Input != 13 || rows[0].Output != 6 || rows[0].CacheRead != 4 || rows[0].CacheWrite != 0 ||
				rows[0].ProviderAccount != "synthetic-go-account" || rows[0].Status != 200 {
				records, _ := json.Marshal(rows)
				test.Errorf("native Go protocol lost its single additive usage owner: %s/stream=%t %s", sample.path, streaming, records)
			} else {
				test.Logf("native Go protocol completed with authoritative additive usage: %s/stream=%t input=13 cached=4 output=6", sample.path, streaming)
			}
			index++
		}
	}
	ctx, attempt := provider.WithExtensionAttempt(test.Context())
	query, err := parseChat([]byte(fmt.Sprintf(cases[0].body, "google/gemini-fixture", true)))
	if err != nil {
		test.Fatal(err)
	}
	events, status, message := server.askTranslated(accounts[0], provider.Chat, "google/gemini-fixture", http.Header{}, http.Header{})(ctx, query)
	if events == nil {
		test.Fatalf("installed search round did not start: %d %s", status, message)
	}
	var text strings.Builder
	for event := range events {
		if event.Kind == KError {
			test.Errorf("installed search round failed: %s", event.Text)
		}
		if event.Kind == KText {
			text.WriteString(event.Text)
		}
	}
	module, operation := attempt.Operation()
	receipt, err := manager.Store().Operation(module, operation)
	totals := attempt.UsageFor(accounts[0])
	if err != nil || receipt.State != "completed" || attempt.Outcome(false) != nil || totals == nil ||
		totals.InputTokens != 13 || totals.CachedTokens != 4 || totals.OutputTokens != 6 || text.String() != "firstsecond" {
		test.Fatalf("installed search round lost its single SDK receipt: receipt=%+v usage=%+v outcome=%v text=%q error=%v",
			receipt, totals, attempt.Outcome(false), text.String(), err)
	}
	test.Log("installed ordinary search round retained its completed receipt and additive usage")
}
